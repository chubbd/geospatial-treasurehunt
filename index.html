First query
-- Select only useful fields so the table and map popup stay readable.
-- name labels dots; lat/lon locate them. id identifies the source record.
SELECT id, name, category, lat, lon
FROM london.places
-- Missing names are NULL; exclude them for this first named-place preview.
WHERE name IS NOT NULL
-- Without ORDER BY these are any 20 matches, not a ranking.
LIMIT 20;
Search places by category
-- Category describes WHAT a place is, rather than words in its name.
-- Replace coffee_shop with restaurant, museum, or another category.
SELECT id, name, category, lat, lon
FROM london.places
-- hierarchy is a list from broad to specific classifications.
-- Search the list to include more specific descendants too.
WHERE list_contains(taxonomy.hierarchy, 'coffee_shop')
-- Alphabetical ordering makes the preview easier to browse.
ORDER BY name NULLS LAST, id
LIMIT 30;
Search places by name
-- name is the dashboard alias for the original names.primary field.
SELECT id, name, category, lat, lon
FROM london.places
-- ILIKE ignores case; % matches any number of characters.
-- '%coffee%' matches anywhere; 'coffee%' matches only the start.
-- Use name = 'Example Cafe' for an exact, case-sensitive match.
WHERE name ILIKE '%coffee%'
ORDER BY name, id
-- A preview, not necessarily every match.
LIMIT 30;
Search places by address
-- A place can have several addresses stored as a list of structs.
-- p aliases the place table; addr represents one address record.
SELECT p.id, p.name, addr.freeform, addr.postcode, p.lat, p.lon
FROM london.places p
-- Expand the list into one row per address, not just the first address.
-- Places with several addresses can repeat; empty address lists disappear.
CROSS JOIN UNNEST(p.addresses) AS t(addr)
-- Search recorded freeform text; % permits text before/after the street.
WHERE addr.freeform ILIKE '%High Street%'
ORDER BY p.name, addr.postcode, p.id
LIMIT 30;
-- Address coverage varies. No match does not prove the place is absent.
-- Try a postcode or name search if freeform is missing.
Search places by postcode
-- Expand all addresses so every recorded postcode can be searched.
SELECT p.id, p.name, addr.freeform, addr.postcode, p.lat, p.lon
FROM london.places p
CROSS JOIN UNNEST(p.addresses) AS t(addr)
-- EC1% means starts with EC1; % is a wildcard and ILIKE ignores case.
-- Use equality instead for an exact full postcode.
WHERE addr.postcode ILIKE 'EC1%'
ORDER BY addr.postcode, p.name, p.id
LIMIT 30;
-- Several addresses can produce several rows for the same place.
Filter buildings
-- Buildings are footprints, not businesses from the places table.
SELECT id, COALESCE(name, 'Building') AS name,
       height, num_floors, subtype, lat, lon
FROM london.buildings
-- AND requires both clues. Unknown (NULL) floor counts do not match 8.
WHERE subtype = 'residential' AND num_floors = 8
ORDER BY id
LIMIT 30;
-- COALESCE supplies a label when name is missing.
-- lat/lon are bbox centres for plotting, not footprint edges.
Distance between points
-- Ellipsoidal point-to-point distance in metres.
-- Spheroid functions expect LATITUDE, LONGITUDE, unlike ST_Point(lon,lat).
SELECT round(ST_Distance_Spheroid(
    ST_Point2D(51.5045, -0.0866),
    ST_Point2D(51.5055, -0.0754)
), 1) AS distance_m;
-- round changes display precision only.
-- No lat/lon columns are returned, so this is a number rather than map dots.
Places within 1 km
-- Find place points within 1000 metres of the specified point.
SELECT id, name, category, lat, lon,
       round(ST_Distance_Spheroid(ST_Point2D(lat, lon),
           ST_Point2D(51.5138, -0.0984)), 1) AS distance_m
FROM london.places
-- Generous rectangular prefilter reduces candidate points.
-- These bounds are for THIS target; update them when moving the target.
WHERE lat BETWEEN 51.5038 AND 51.5238
  AND lon BETWEEN -0.1154 AND -0.0814
  -- Actual ellipsoidal radius test, not a rectangle or rounded distance.
  AND ST_DWithin_Spheroid(ST_Point2D(lat, lon),
      ST_Point2D(51.5138, -0.0984), 1000)
-- Rank by unrounded distance to avoid rounding ties changing the order.
ORDER BY ST_Distance_Spheroid(ST_Point2D(lat, lon),
    ST_Point2D(51.5138, -0.0984)), id
LIMIT 30;
Closest building centres
-- This ranks representative CENTRES, not the nearest footprint edge.
SELECT id, COALESCE(name, 'Building') AS name, lat, lon,
       round(ST_Distance_Spheroid(ST_Point2D(lat, lon),
           ST_Point2D(51.5074, -0.1278)), 1) AS distance_m
FROM london.buildings
-- Only known coordinates can be compared.
WHERE lat IS NOT NULL AND lon IS NOT NULL
-- Sort before LIMIT so these are the closest 15, not any 15.
ORDER BY ST_Distance_Spheroid(ST_Point2D(lat, lon),
    ST_Point2D(51.5074, -0.1278)), id
LIMIT 15;
-- Ranking all London buildings can be slower than a local search.
-- Use the footprint example if ANY PART of a building must be in range.
Footprint within 500 m
-- Question: is ANY PART of a building within 500 m of this point?
-- Using its centre instead would answer a different question.
WITH target AS (
    -- Ordinary ST_Point uses LONGITUDE, LATITUDE.
    -- WGS84 is in degrees; project to British National Grid for metres.
    -- always_xy preserves the longitude/latitude input order.
    SELECT ST_Transform(ST_Point(-0.0984, 51.5138),
        'EPSG:4326', 'EPSG:27700', always_xy := true) AS geom_m
), buildings AS (
    -- WITH names intermediate queries so each step is easier to follow.
    -- Both geometries must use the SAME projected coordinate system.
    SELECT id, name, lat, lon, ST_Transform(geometry,
        'EPSG:4326', 'EPSG:27700', always_xy := true) AS geom_m
    FROM london.buildings
    -- Cheap rectangular prefilter before the polygon-distance test.
    -- Compare bbox EDGES so crossing footprints are not excluded.
    -- Update this generous envelope if you change the target/radius.
    WHERE bbox.xmax >= -0.1104 AND bbox.xmin <= -0.0864
      AND bbox.ymax >= 51.5068 AND bbox.ymin <= 51.5208
)
SELECT b.id, COALESCE(b.name, 'Building') AS name, b.lat, b.lon,
       -- Distance is to the nearest footprint part; zero if point is inside.
       round(ST_Distance(b.geom_m, t.geom_m), 1) AS distance_m
FROM buildings b
-- target has ONE row; pair each building with that one target point.
CROSS JOIN target t
-- Actual 500 m test uses unrounded distances in projected metres.
WHERE ST_DWithin(b.geom_m, t.geom_m, 500)
ORDER BY ST_Distance(b.geom_m, t.geom_m), b.id
LIMIT 30;
-- Map dots use centres even though filtering uses footprint edges.
-- Spherical/spheroidal functions take points, not building polygons.
-- Keep geometry inside calculations; return readable map properties.
Measured footprint area
-- Measure the polygon, not the dashboard's approximate bbox area.
SELECT id, COALESCE(name, 'Building') AS name, lat, lon,
       -- Project to metres before measuring: area then uses square metres.
       round(ST_Area(ST_Transform(geometry,
           'EPSG:4326', 'EPSG:27700', always_xy := true
       )), 1) AS footprint_area_m2
FROM london.buildings
-- Narrow candidates before expensive polygon calculations.
WHERE subtype = 'residential' AND num_floors = 8
ORDER BY id
LIMIT 30;
-- External footprint area is not total floor area or internal usable space.
Building contains a point
-- Find footprints whose INTERIOR contains the supplied point.
SELECT id, COALESCE(name, 'Building') AS name, lat, lon
FROM london.buildings
-- Both inputs are WGS84. This tests topology, not metre distances.
WHERE ST_Contains(geometry, ST_Point(-0.0984, 51.5138))
ORDER BY id
LIMIT 10;
-- A point exactly on the boundary does not pass ST_Contains.
-- Use ST_Intersects instead if boundary points should count.

Tables & schema

Places: businesses and amenities represented by points. Buildings: physical footprint polygons. IDs do not directly join across these tables.

name aliases names.primary. category aliases taxonomy.primary with basic_category fallback. taxonomy.hierarchy is a list: list_contains searches its classifications. addresses is a list of structs: UNNEST expands each address.

lat/lon are dashboard-added bbox centres. building_area_m2 is an approximate bbox area, not measured footprint area. num_floors is a recorded storey count; height is in metres. NULL means unknown.
